<script type="text/x-dc" data-dc-script>
class Component extends DCLogic {
  state = { f: null, ready: !!window.SITE };
  componentDidMount() {
    if (!window.SITE) this._t = setInterval(() => { if (window.SITE) { clearInterval(this._t); this.setState({ ready: true }); } }, 30);
  }
  componentWillUnmount() { clearInterval(this._t); }
  pick(f) {
    const search = f ? '?f=' + f : '';
    if (this.props.setquery) return this.props.setquery(search);
    try { history.replaceState(null, '', search || location.pathname); } catch (_) {}
    this.setState({ f });
  }
  renderVals() {
    const S = window.SITE;
    if (!S) return {};
    let q = null; try { q = new URLSearchParams(this.props.currentquery ?? location.search).get('f'); } catch (_) {}
    const f = this.state.f !== null ? this.state.f : (S.services.some(s => s.slug === q) ? q : '');
    const name = Object.fromEntries(S.services.map(s => [s.slug, s.name]));
    const list = S.works.filter(w => !f || w.dirs.includes(f));
    const opts = [{ slug: '', label: 'Все работы' }, ...S.services.map(s => ({ slug: s.slug, label: s.name }))];
    return {
      filters: opts.map(o => {
        const on = o.slug === f;
        return { label: o.label, count: o.slug ? S.works.filter(w => w.dirs.includes(o.slug)).length : S.works.length, sel: on ? 'true' : 'false', bg: on ? '#171512' : 'transparent', fg: on ? '#F3F0EA' : '#171512', pick: () => this.pick(o.slug) };
      }),
      items: list.map((work, i) => ({
        ...S.workCard(work),
        n: String(i + 1).padStart(2, '0'),
        dirNames: [work.kind, ...work.dirs.map(d => name[d])].filter(Boolean).join(' · ')
      }))
    };
  }
}
</script>
