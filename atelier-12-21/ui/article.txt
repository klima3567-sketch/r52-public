<script type="text/x-dc" data-dc-script data-props="{&quot;article&quot;:{&quot;editor&quot;:&quot;enum&quot;,&quot;options&quot;:[&quot;kak-vybrat-tkan&quot;,&quot;chto-proverit-na-primerke&quot;,&quot;s-chego-nachat-kapsulu&quot;],&quot;default&quot;:&quot;kak-vybrat-tkan&quot;,&quot;tsType&quot;:&quot;string&quot;}}">
class Component extends DCLogic {
  state = { ready: !!window.SITE, wide: true };
  componentDidMount() {
    this.onR = () => { const w = window.innerWidth >= 900; if (w !== this.state.wide) this.setState({ wide: w }); };
    this.onR(); window.addEventListener('resize', this.onR);
    if (!window.SITE) this._t = setInterval(() => { if (window.SITE) { clearInterval(this._t); this.setState({ ready: true }); } }, 30);
  }
  componentWillUnmount() { clearInterval(this._t); window.removeEventListener('resize', this.onR); }
  renderVals() {
    const S = window.SITE;
    if (!S) return { ready: false, a: {}, asidePos: 'static' };
    let q = null; try { q = new URLSearchParams(this.props.currentquery ?? location.search).get('a'); } catch (_) {}
    const a = S.articles.find(x => x.slug === (q || this.props.article)) || S.articles[0];
    const dir = S.services.find(s => s.slug === a.dir);
    return {
      ready: true, a, asidePos: this.state.wide ? 'sticky' : 'static',
      toc: a.sections.map((s, i) => ({ ...s, n: String(i + 1).padStart(2, '0'), id: 's' + (i + 1), href: '#s' + (i + 1) })),
      dirHref: 'directions.html?d=' + dir.slug, dirName: dir.name,
      more: S.articles.filter(x => x.slug !== a.slug).map(x => ({ ...x, href: 'article.html?a=' + x.slug }))
    };
  }
}
</script>
