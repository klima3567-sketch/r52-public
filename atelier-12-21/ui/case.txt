class Component extends DCLogic {
  state = { ready: !!window.SITE, wide: true };
  componentDidMount() {
    this.onR = () => { const w = window.innerWidth >= 900; if (w !== this.state.wide) this.setState({ wide: w }); };
    this.onR(); window.addEventListener('resize', this.onR);
    if (!window.SITE) this._t = setInterval(() => { if (window.SITE) { clearInterval(this._t); this.setState({ ready: true }); } }, 30);
  }
  componentWillUnmount() { clearInterval(this._t); window.removeEventListener('resize', this.onR); }
  renderVals() {
    const S = window.SITE;
    if (!S) return { ready: false, w: {}, prev: {}, next: {}, colPos: 'static' };
    let q = null; try { q = new URLSearchParams(location.search).get('w'); } catch (_) {}
    const ws = S.works;
    let i = ws.findIndex(x => x.id === (this.props.selected || q || this.props.work));
    if (i < 0) i = 0;
    const w = ws[i];
    const svc = Object.fromEntries(S.services.map(s => [s.slug, s]));
    const link = S.workCard;
    const plates = w.plates || [];
    const cover = { img: w.img, alt: w.alt || w.title, label: w.viewLabel || 'Общий вид' };
    const photos = [cover, ...(w.additionalViews || [])];
    const current = this.state.viewWork === w.id ? this.state.viewIndex : 0;
    const views = photos.map((photo, index) => ({
      ...photo,
      selected: index === current ? 'true' : 'false',
      border: index === current ? '#171512' : '#B9B5AD',
      background: index === current ? '#E8E4DD' : 'transparent',
      pick: () => this.setState({ viewWork: w.id, viewIndex: index })
    }));
    const fields = w.details || [
      { label: 'Задача', value: w.task },
      { label: 'Решение', value: w.solution },
      { label: 'Ткань и фурнитура', value: w.fabric },
      { label: 'Сроки и примерки', value: w.timing }
    ].filter(field => field.value);
    const caseNote = plates.length
      ? 'Каждое фото показывает отдельное изделие. Для вашей капсулы совместимость цветов, тканей и вещей подберём на встрече.'
      : w.kind === 'Визуализация модели · ИИ' ? 'Пример возможного пошива. Изображение создано с помощью ИИ. Ткань, конструкцию и сроки обсудим на консультации.'
      : w.kind ? 'По этой модели можно обсудить силуэт и детали своего заказа.' : 'Изделие из портфолио ателье. Цвет, силуэт и сочетание деталей нового изделия обсуждаем индивидуально.';
    const main = svc[w.dirs[0]];
    return {
      ready: true, w, colPos: this.state.wide ? 'sticky' : 'static',
      ratio: w.ratio || '4 / 5',
      pos: w.imgPos || 'center top',
      dirs: w.dirs.map(d => ({ name: svc[d].name, href: 'directions.html?d=' + d })),
      dirHref: 'directions.html?d=' + main.slug,
      dirName: main.name,
      fields, plates, caseNote: w.photoNote || caseNote,
      photo: photos[current] || cover, views, hasAdditionalViews: photos.length > 1,
      portfolioHref: w.photoNote ? 'portfolio.html?f=' + main.slug : 'portfolio.html',
      hasKind: !!w.kind, hasPlates: !!plates.length, hasSinglePhoto: !plates.length,
      prev: link(ws[(i - 1 + ws.length) % ws.length]),
      next: link(ws[(i + 1) % ws.length]),
      more: ws.filter(x => x.id !== w.id).sort((a, b) => (b.dirs.some(d => w.dirs.includes(d)) ? 1 : 0) - (a.dirs.some(d => w.dirs.includes(d)) ? 1 : 0)).slice(0, 4).map(link)
    };
  }
}
